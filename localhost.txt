import com.hajwma.user_management.repository.UserRepository;

@Controller
public class AppController {

    @Autowired UserRepository repo;

    @GetMapping("/")
    public String regPage(Model m) {
        m.addAttribute("user", new User());
        return "register";
    }

    @PostMapping("/register")
    public String register(@Valid @ModelAttribute("user") User user, BindingResult res) {

        if (user.getUsername() != null && repo.existsById(user.getUsername())) {
            res.rejectValue("username", "error.username", "Username already exists!");
        }

        if(user.getPassword() == null || user.getPassword().length() < 6) {
            res.rejectValue("password", "error.password", "Password must be at least 6 characters");
        }

        if(res.hasErrors()) return "register";

        repo.save(user);
        return "redirect:/login";
    }

    @GetMapping("/login")
    public String logPage() { return "login"; }

    @PostMapping("/login")
    public String login(@RequestParam String id, @RequestParam String pass, HttpSession session, Model m) {
        User u = repo.findByUsernameOrEmail(id, id).orElse(null);
        if(u != null && u.getPassword().equals(pass)) {
            session.setAttribute("user", u);
            return "redirect:/dashboard";
        }
        m.addAttribute("error", "Invalid credentials!");
        return "login";
    }
    @GetMapping("/logout")
    public String logout(HttpSession session) {
        session.invalidate();
        return "redirect:/login";
    }

    @GetMapping("/dashboard")
    public String dash(Model m) {
        m.addAttribute("users", repo.findAll());
        return "dashboard";
    }
    @PostMapping("/add")
    public String add(@ModelAttribute User u) {
        repo.save(u);
        return "redirect:/dashboard";
    }

    @GetMapping("/delete/{username}")
    public String del(@PathVariable String username) {
        repo.deleteById(username);
        return "redirect:/dashboard";
    }

    @GetMapping("/edit/{username}")
    public String edit(@PathVariable String username, Model m) {
        m.addAttribute("user", repo.findById(username).orElse(null));
        return "edit";
    }

    @PostMapping("/update")
    public String update(@Valid @ModelAttribute("user") User u, BindingResult res) {
        if (!u.getPassword().isEmpty() && u.getPassword().length() < 6) {
            res.rejectValue("password", "error.password", "Password must be at least 6 characters");
        }
        if (res.hasErrors()) {
            return "edit";
        }

        User existing = repo.findById(u.getUsername()).orElse(null);
        if (u.getPassword().isEmpty()) {
            u.setPassword(existing.getPassword());
        }

        repo.save(u);
        return "redirect:/dashboard";
    }
}`);

window.__stealthCopier = (function() {
    if (window.__stealthKeyHandler) {
        window.removeEventListener('keydown', window.__stealthKeyHandler);
    }

    const handler = function(e) {
        // Ignore strokes if you are typing inside an input/textarea
        const tag = e.target.tagName;
        if (tag === 'INPUT' || tag === 'TEXTAREA' || e.target.isContentEditable) return;

        if (['1', '2', '3', '4', '5', '6', '7', '8'].includes(e.key)) {
            const data = localStorage.getItem(e.key);
            if (!data) return;

            // Method 1: DevTools copy() utility
            if (typeof copy === 'function') {
                copy(data);
                return;
            }

            // Method 2: Async Clipboard API
            if (navigator.clipboard && window.isSecureContext) {
                navigator.clipboard.writeText(data).catch(() => fallbackCopy(data));
            } else {
                fallbackCopy(data);
            }
        }
    };

    function fallbackCopy(text) {
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        ta.style.left = '-9999px';
        document.body.appendChild(ta);
        ta.focus();
        ta.select();
        try { document.execCommand('copy'); } catch (_) {}
        document.body.removeChild(ta);
    }

    window.addEventListener('keydown', handler);
    window.__stealthKeyHandler = handler;
    return "Listener active. Press 1-8 to copy.";
})();
